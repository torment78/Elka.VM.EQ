cmake_minimum_required(VERSION 3.22)
project(ElkaVmEqNative VERSION 0.1.0 LANGUAGES C RC)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(eq_core STATIC src/eq_core.c)
add_library(spectrum STATIC src/spectrum.c)
target_include_directories(spectrum PUBLIC src)
target_compile_definitions(spectrum PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(eq_core PUBLIC src)
add_executable(ElkaVmEq WIN32 src/main.c src/vm_api.c src/app.rc)
target_link_libraries(ElkaVmEq PRIVATE eq_core spectrum user32 gdi32 msimg32 advapi32 shell32 comdlg32)
target_compile_definitions(ElkaVmEq PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_definitions(eq_core PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
 target_link_options(ElkaVmEq PRIVATE /MANIFEST:NO)
 target_compile_options(eq_core PRIVATE /W4 /utf-8)
 target_compile_options(ElkaVmEq PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(EqChecks tests/eq_checks.c)
target_link_libraries(EqChecks PRIVATE eq_core)
add_test(NAME EqCore COMMAND EqChecks)
add_executable(VmChecks tests/vm_checks.c src/vm_api.c)
target_link_libraries(VmChecks PRIVATE eq_core spectrum advapi32)
target_compile_definitions(VmChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_test(NAME VmQueue COMMAND VmChecks)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT ElkaVmEq)
add_library(ElkaEqEngine SHARED src/eq_bridge.c src/vm_api.c)
target_link_libraries(ElkaEqEngine PRIVATE eq_core spectrum advapi32)
target_compile_definitions(ElkaEqEngine PRIVATE EQ_ENGINE_BUILD UNICODE _UNICODE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
if(MSVC)
 target_compile_options(ElkaEqEngine PRIVATE /W4 /utf-8)
endif()
add_executable(BridgeChecks tests/bridge_checks.c)
target_link_libraries(BridgeChecks PRIVATE ElkaEqEngine)
target_include_directories(BridgeChecks PRIVATE src)
add_test(NAME CAbiBridge COMMAND BridgeChecks)
add_executable(SpectrumChecks tests/spectrum_checks.c)
target_link_libraries(SpectrumChecks PRIVATE spectrum)
target_compile_definitions(SpectrumChecks PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME Spectrum COMMAND SpectrumChecks)
